def predict(X: dict):
    # Initialize the list to track the truth values of the inner nodes
    nodes = 2 * [None]

    # Check the first condition based on social_connection_frequency
    nodes[0] = X["social_connection_frequency"] in ["Weekly", "Never", "Monthly"]
    
    if nodes[0]:
        # Prediction for those who rarely keep in touch or maintain friendships through calls
        prediction = 1  # Target: 1 means maintains long-distance friendships through calls
    else:
        # Check the second condition based on social_orientation_type
        nodes[1] = X["social_orientation_type"] == "Extrovert"
        
        if nodes[1]:
            # Prediction for those who use texting/social media
            prediction = 0  # Target: 0 means uses texting/social media
        else:
            # If not extrovert, they also maintain long-distance friendships through calls
            prediction = 1  # Target: 1 means maintains long-distance friendships through calls

    return prediction, nodes

# Example usage:
