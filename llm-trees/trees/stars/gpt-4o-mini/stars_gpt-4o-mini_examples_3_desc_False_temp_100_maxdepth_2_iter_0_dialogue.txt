```python
def predict(X: dict):

    # First decision node: Check if B-V <= 0.5
    nodes[0] = X["B-V"] <= 0.5
    
    if nodes[0]:
        # If the condition is met, classify as Dwarf Star
        prediction = 0  # 0 = Dwarf Star
    else:
        # Second decision node: Check if Vmag <= 15
        nodes[1] = X["Vmag"] <= 15
        
        if nodes[1]:

    return prediction, nodes
```
